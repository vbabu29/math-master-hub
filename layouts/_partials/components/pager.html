{{/* Article navigation — uses shortTitle when set to avoid awkward wrapping */}} {{- $reversePagination := .Store.Get "reversePagination" | default false -}} {{- $prev := cond $reversePagination .PrevInSection .NextInSection -}} {{- $next := cond $reversePagination .NextInSection .PrevInSection -}} {{- if eq .Params.prev false }} {{- if $reversePagination }}{{ $next = false }}{{ else }}{{ $prev = false }}{{ end -}} {{ else }} {{- with .Params.prev -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $next = . }}{{ else }}{{ $prev = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if eq .Params.next false }} {{- if $reversePagination }}{{ $prev = false }}{{ else }}{{ $next = false }}{{ end -}} {{ else }} {{- with .Params.next -}} {{- with $.Site.GetPage . -}} {{- if $reversePagination }}{{ $prev = . }}{{ else }}{{ $next = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $prev $next -}}
{{- if $prev -}} {{- $fullTitle := partial "utils/title" $prev -}} {{- $shortTitle := $prev.Params.shortTitle | default $fullTitle -}} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"apex-pager-icon apex-pager-icon-prev\"") -}} Previous {{ $shortTitle }} {{- else -}} {{- end -}} {{- if $next -}} {{- $fullTitle := partial "utils/title" $next -}} {{- $shortTitle := $next.Params.shortTitle | default $fullTitle -}} Next {{ $shortTitle }} {{- partial "utils/icon.html" (dict "name" "chevron-right" "attributes" "class=\"apex-pager-icon\"") -}} {{- end -}}
{{- end -}}
